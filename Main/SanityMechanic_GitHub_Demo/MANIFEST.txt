--------------------------------------------
Content.Server/Administration/Systems/AdminVerbSystem.Smites.cs
--------------------------------------------
1) Add these usings at the top:

using Content.Server._Sunrise.Sanity;
using Content.Shared._Sunrise.Sanity;
using Content.Shared.Humanoid;

2) Add these two dependencies next to the other [Dependency] fields:

    [Dependency] private readonly SanitySystem _sanitySystem = default!;
    [Dependency] private readonly HallucinationSystem _hallucinationSystem = default!;

3) Add this block anywhere inside AddSmiteVerbs(), e.g. right after
   args.Verbs.Add(homingRodSlow);:

        if (TryComp<SanityComponent>(args.Target, out var sanityComp))
        {
            var lowerSanityName = Loc.GetString("admin-smite-lower-sanity-name").ToLowerInvariant();
            Verb lowerSanity = new()
            {
                Text = lowerSanityName,
                Category = VerbCategory.Smite,
                Icon = new SpriteSpecifier.Texture(new("/Textures/Interface/VerbIcons/smite.svg.192dpi.png")),
                Act = () =>
                {
                    _sanitySystem.LowerThresholdByOne(args.Target, sanityComp);
                },
                Impact = LogImpact.Medium,
                Message = string.Join(": ", lowerSanityName, Loc.GetString("admin-smite-lower-sanity-description"))
            };
            args.Verbs.Add(lowerSanity);

            var viewSanityName = Loc.GetString("admin-smite-view-sanity-name").ToLowerInvariant();
            Verb viewSanity = new()
            {
                Text = viewSanityName,
                Category = VerbCategory.Debug,
                Icon = new SpriteSpecifier.Texture(new("/Textures/Interface/VerbIcons/information.svg.192dpi.png")),
                Act = () =>
                {
                    var value = MathF.Round(sanityComp.CurrentSanity, 1);
                    var immuneNote = sanityComp.IsMedicalImmune ? " [мед. иммунитет]" : "";
                    _popup.PopupEntity($"Психика: {value}% ({sanityComp.CurrentThreshold}){immuneNote}", args.Target, args.User);
                },
                Impact = LogImpact.Low,
                Message = string.Join(": ", viewSanityName, Loc.GetString("admin-smite-view-sanity-description"))
            };
            args.Verbs.Add(viewSanity);

            var restoreSanityName = Loc.GetString("admin-smite-restore-sanity-name").ToLowerInvariant();
            Verb restoreSanity = new()
            {
                Text = restoreSanityName,
                Category = VerbCategory.Tricks,
                Icon = new SpriteSpecifier.Texture(new("/Textures/Interface/AdminActions/rejuvenate.png")),
                Act = () =>
                {
                    _sanitySystem.RestoreSanity(args.Target, sanityComp);
                },
                Impact = LogImpact.Medium,
                Message = string.Join(": ", restoreSanityName, Loc.GetString("admin-smite-restore-sanity-description"))
            };
            args.Verbs.Add(restoreSanity);
        }

        if (HasComp<HumanoidAppearanceComponent>(args.Target))
        {
            var summonHallucinationName = Loc.GetString("admin-smite-summon-hallucination-name").ToLowerInvariant();
            Verb summonHallucination = new()
            {
                Text = summonHallucinationName,
                Category = VerbCategory.Smite,
                Icon = new SpriteSpecifier.Texture(new("/Textures/Interface/VerbIcons/smite.svg.192dpi.png")),
                Act = () =>
                {
                    _hallucinationSystem.SpawnHallucinationFor(args.Target, args.Target);
                },
                Impact = LogImpact.Medium,
                Message = string.Join(": ", summonHallucinationName, Loc.GetString("admin-smite-summon-hallucination-description"))
            };
            args.Verbs.Add(summonHallucination);
        }
